### 0.5 【重要】对设计文档的有意偏离（必须记入计划，不要"纠正"回来）

设计文档 §3 写的是 **Room（收藏/历史/歌词缓存）+ DataStore（设置）**。本里程碑**主动改为**：

- 收藏 / 历史 / 歌词磁盘缓存 → **`JsonFileStore`（Gson 原子写 JSON 文件，存于 `context.filesDir`）**
- 设置 / 登录会话 → **`SharedPreferences`（经 `KeyValueStore` 接口抽象）**

**理由：** Room 需要 **KSP**（annotation processor）。本项目 Kotlin 为 **2.4.21**，而本机可用的最新 KSP（`2.3.12`）**不覆盖该 Kotlin 版本**，强行引入会导致构建直接失败（高风险的版本地狱）。DataStore 则会额外引入 `androidx.datastore` 依赖与其协程/序列化配置成本。改用「纯 Kotlin 仓储 + Gson + SharedPreferences」后：

- **零新增构建插件**，构建风险最低；
- 全部核心逻辑是**纯 Kotlin**，可在 `:app:testDebugUnitTest` 里完整单测（满足本项目的测试门禁）；
- 数据量极小（个人自用、单机收藏/历史/若干首歌的歌词），JSON 文件 + LRU 淘汰完全够用。

**这与设计文档的取舍不同，属有意为之；spec §3 的技术选型以本计划为准。**

## 1. 已核实的接口事实（**一手实测，不要凭记忆改动**）

### 1.1 扫码登录（复用 eapi，**不需要 weapi / RSA**）

| 步骤 | 请求 |
|---|---|
| 取二维码 key | `POST https://interfacepc.music.163.com/eapi/login/qrcode/unikey`，uri 用 `/api/login/qrcode/unikey`，payload `{"type":1,"e_r":false,"header":{...}}` → `{"code":200,"unikey":"<uuid>"}` |
| 二维码内容 | `https://music.163.com/login?codekey=<unikey>`（**本地用 ZXing 渲染成位图，不加载网页**） |
| 轮询 | 同 host `/eapi/login/qrcode/client/login`，payload `{"key":"<unikey>","type":1,"e_r":false,"header":{...}}` |

轮询 `code` 语义（实测）：**`801` 等待扫码 / `800` 二维码过期 / `802` 已扫码待确认 / `803` 登录成功**。`803` 的响应体里带 `cookie` 字符串，内含 **`MUSIC_U`** 与 **`__csrf`**。

取昵称（可选）：`POST .../eapi/nuser/account/get`，uri `/api/nuser/account/get`，payload `{"e_r":false,"header":{...}}`，Cookie 需带 `MUSIC_U`；响应 `{"code":200,"profile":{"nickname":"..."}}`。**失败不影响登录成功**（昵称可空）。

### 1.2 备用源协议（NetEaseCloudMusicApi 风格）

默认地址**留空**（= 未配置，不参与正常流程）。配置后走标准 REST：

| 能力 | 请求 | 响应（与本项目 `NetEaseDto` 完全一致，可直接复用） |
|---|---|---|
| 搜索 | `GET {base}/search?keywords=<urlenc>&type=1&limit=30&offset=<n>` | `{"code":200,"result":{"songs":[...]}}` |
| 取流 | `GET {base}/song/url/v1?id=<id>&level=exhigh` | `{"code":200,"data":[{...,"freeTrialInfo":{...}}]}` |
| 歌词 | `GET {base}/lyric?id=<id>` | `{"code":200,"lrc":{"lyric":..},"yrc":{..},"tlyric":{..},"romalrc":{..}}` |

> `type` 参数直接复用 `SearchType.apiValue`（SONG=1 / ALBUM=10 / ARTIST=100）。

```kotlin
package com.musicbox.app.data

import android.content.SharedPreferences

/**
 * 键值存储抽象。抽出来的唯一目的是让 SettingsRepository / SessionStore 能在
 * JVM 单测里跑（真机用 SharedPreferences，单测用内存实现）。
 */
interface KeyValueStore {
    fun getString(key: String): String?
    fun putString(key: String, value: String?)
    fun getBoolean(key: String, defaultValue: Boolean): Boolean
    fun putBoolean(key: String, value: Boolean)
    fun remove(key: String)
}

/** 真机实现：应用私有 SharedPreferences。 */
class SharedPrefsKeyValueStore(private val prefs: SharedPreferences) : KeyValueStore {
    override fun getString(key: String): String? = prefs.getString(key, null)
    override fun putString(key: String, value: String?) {
        prefs.edit().putString(key, value).apply()
    }
    override fun getBoolean(key: String, defaultValue: Boolean): Boolean =
        prefs.getBoolean(key, defaultValue)
    override fun putBoolean(key: String, value: Boolean) {
        prefs.edit().putBoolean(key, value).apply()
    }
    override fun remove(key: String) {
        prefs.edit().remove(key).apply()
    }
}
```

- [ ] **Step 2: 写 `JsonFileStore.kt`**

```kotlin
package com.musicbox.app.data

import com.google.gson.Gson
import java.io.File
import java.lang.reflect.Type

/**
 * 泛型 JSON 文件存储。原子写：先写 `<name>.tmp`，再 rename 覆盖目标，
 * 避免"写到一半进程被杀"留下半个损坏文件。
 *
 * 读失败（文件不存在 / 内容损坏）一律返回 null，由调用方当作"空"处理。
 * 纯 JVM 实现，单测可注入临时目录的 File。
 */
class JsonFileStore<T : Any>(
    private val file: File,
    private val type: Type,
    private val gson: Gson = Gson(),
) {

    fun read(): T? {
        if (!file.exists()) return null
        return try {
            gson.fromJson(file.readText(), type)
        } catch (e: Exception) {
            null
        }
    }

    fun write(value: T) {
        file.parentFile?.mkdirs()
        val tmp = File(file.parentFile, file.name + TMP_SUFFIX)
        tmp.writeText(gson.toJson(value, type))
        if (file.exists() && !file.delete()) {
            // 删不掉就直接覆盖（copyTo overwrite 会先清空），不做静默失败
            tmp.copyTo(file, overwrite = true)
            tmp.delete()
            return
        }
        if (!tmp.renameTo(file)) {
            tmp.copyTo(file, overwrite = true)
            tmp.delete()
        }
    }

    fun delete() {
        if (file.exists()) file.delete()
    }

    fun exists(): Boolean = file.exists()

    private companion object {
        const val TMP_SUFFIX = ".tmp"
    }
}
```

```kotlin
package com.musicbox.app.data

import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow

/** 登录会话的持久化。写入即更新 StateFlow，UI 与音源层共读一份。 */
class SessionStore(private val store: KeyValueStore) {

    private val _session = MutableStateFlow(read())
    val session: StateFlow<NetEaseSession> = _session.asStateFlow()

    fun save(musicU: String, csrf: String, nickname: String?) {
        store.putString(KEY_MUSIC_U, musicU)
        store.putString(KEY_CSRF, csrf)
        store.putString(KEY_NICKNAME, nickname)
        _session.value = NetEaseSession(musicU, csrf, nickname)
    }

    /** 昵称是"可选增强"：拿到了就补，拿不到不影响登录态。 */
    fun updateNickname(nickname: String?) {
        if (nickname.isNullOrBlank()) return
        store.putString(KEY_NICKNAME, nickname)
        _session.value = _session.value.copy(nickname = nickname)
    }

    fun clear() {
        store.remove(KEY_MUSIC_U)
        store.remove(KEY_CSRF)
        store.remove(KEY_NICKNAME)
        _session.value = NetEaseSession.Guest
    }

    private fun read(): NetEaseSession = NetEaseSession(
        musicU = store.getString(KEY_MUSIC_U)?.takeIf { it.isNotBlank() },
        csrf = store.getString(KEY_CSRF).orEmpty(),
        nickname = store.getString(KEY_NICKNAME),
    )

    private companion object {
        const val KEY_MUSIC_U = "login_music_u"
        const val KEY_CSRF = "login_csrf"
        const val KEY_NICKNAME = "login_nickname"
    }
}
```

```kotlin
import com.google.gson.reflect.TypeToken
import com.musicbox.app.model.Song

import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow

/** 一条收藏记录。带上收藏时间是为了排序稳定（新的在前）。 */
data class FavoriteEntry(val song: Song, val addedAtMs: Long)

/**
 * 收藏仓储。全量 JSON 落盘（数据量小），每次变更写一次。
 * UI 通过 [favorites] 的 StateFlow 订阅。
 */
class FavoriteRepository(
    private val store: JsonFileStore<List<FavoriteEntry>>,
    private val clock: () -> Long = System::currentTimeMillis,
) {

    private val _favorites = MutableStateFlow(store.read().orEmpty())
    val favorites: StateFlow<List<FavoriteEntry>> = _favorites.asStateFlow()

    fun isFavorite(songId: Long): Boolean = _favorites.value.any { it.song.id == songId }

    /** 收藏 / 取消收藏，返回操作后是否处于"已收藏"。 */
    fun toggle(song: Song): Boolean {
        val current = _favorites.value
        val exists = current.any { it.song.id == song.id }
        val next = if (exists) {
            current.filterNot { it.song.id == song.id }
        } else {
            listOf(FavoriteEntry(song, clock())) + current
        }
        persist(next)
        return !exists
    }

    fun remove(songId: Long) {
        persist(_favorites.value.filterNot { it.song.id == songId })
    }

    private fun persist(list: List<FavoriteEntry>) {
        _favorites.value = list
        store.write(list)
    }

    companion object {
        /** 统一在这里声明 Gson 的泛型 Type，避免各处各写一份。 */
        val listType: java.lang.reflect.Type =
            object : TypeToken<List<FavoriteEntry>>() {}.type

        fun forDir(dir: File): FavoriteRepository =
            FavoriteRepository(JsonFileStore(File(dir, "favorites.json"), listType))
    }
```

```kotlin
package com.musicbox.app.data

import com.google.gson.Gson
import com.musicbox.app.model.LyricData
import java.io.File

/**
 * 歌词磁盘缓存：按 songId 存**原始 LyricData JSON**（不是解析后的时间轴）。
 * 存原始文本的好处是：以后解析器升级（比如支持新格式）时，已缓存数据能直接受益，
 * 不必让用户重新联网拉一遍。读出来再解析，解析是纯 CPU 开销。
 */
class LyricDiskCache(
    private val dir: File,
    private val gson: Gson = Gson(),
) {

    fun get(songId: Long): LyricData? {
        val file = fileFor(songId)
        if (!file.exists()) return null
        return try {
            gson.fromJson(file.readText(), LyricData::class.java)
        } catch (e: Exception) {
            file.delete() // 损坏文件直接丢弃，下次重新拉
            null
        }
    }

    fun put(songId: Long, data: LyricData) {
        dir.mkdirs()
        val target = fileFor(songId)
        val tmp = File(dir, "$songId.json$TMP_SUFFIX")
        tmp.writeText(gson.toJson(data))
        if (target.exists() && !target.delete()) {
            tmp.copyTo(target, overwrite = true)
            tmp.delete()
            return
        }
        if (!tmp.renameTo(target)) {
            tmp.copyTo(target, overwrite = true)
            tmp.delete()
        }
    }

    fun size(): Int = dir.listFiles { f -> f.name.endsWith(".json") }?.size ?: 0

    fun clear() {
        dir.listFiles()?.forEach { it.delete() }
    }

    private fun fileFor(songId: Long) = File(dir, "$songId.json")

    companion object {
        private const val TMP_SUFFIX = ".tmp"

        fun forDir(dir: File): LyricDiskCache = LyricDiskCache(dir)
    }
}
```

```kotlin
package com.musicbox.app.data

import java.io.File
import java.time.Instant

/**
 * 本地滚动日志文件。用途（设计文档 §9）：签名 / 解析异常这类"不该让用户看到"
 * 的技术错误，写进文件供排查；用户侧只看到友好文案。
 *
 * 滚动策略：文件超过 [maxBytes] 时，保留尾部一半内容重写，避免无限增长。
 */
class AppLog(
    private val file: File,
    private val maxBytes: Long = DEFAULT_MAX_BYTES,
    private val clock: () -> Long = System::currentTimeMillis,
) {

    fun log(tag: String, message: String, throwable: Throwable? = null) {
        val timestamp = Instant.ofEpochMilli(clock()).toString()
        val detail = throwable?.let { " | ${it.javaClass.simpleName}: ${it.message}" }.orEmpty()
        append("$timestamp [$tag] $message$detail")
    }

    fun readAll(): String = if (file.exists()) file.readText() else ""

    fun clear() {
        if (file.exists()) file.delete()
    }

    private fun append(line: String) {
        file.parentFile?.mkdirs()
        rollIfNeeded()
        file.appendText(line + "\n")
    }

    private fun rollIfNeeded() {
        if (!file.exists() || file.length() <= maxBytes) return
        val tail = file.readText().takeLast((maxBytes / 2).toInt())
        file.writeText(tail)
    }

    companion object {
        const val DEFAULT_MAX_BYTES: Long = 256 * 1024

        fun forDir(dir: File): AppLog = AppLog(File(dir, "musicbox.log"))
    }
}
```

```kotlin
package com.musicbox.app.ui.lyric

import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import com.musicbox.app.data.LyricDiskCache
import com.musicbox.app.lyric.LyricParser
import com.musicbox.app.lyric.LyricTimeline
import com.musicbox.app.lyric.sharedLyricCache
import com.musicbox.app.source.MusicProvider
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.Job
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.distinctUntilChanged
import kotlinx.coroutines.launch
import kotlinx.coroutines.withContext

/** 歌词页的四种终态 + 两种中间态。 */
sealed interface LyricUiState {
    data object Idle : LyricUiState
    data object Loading : LyricUiState
    data class Ready(val timeline: LyricTimeline) : LyricUiState
    data object Empty : LyricUiState
    /** 拉取失败。UI 一律显示"暂无歌词"，绝不影响播放（设计文档 §9）。 */
    data object Failed : LyricUiState
}

/**
 * 歌词状态机：切歌 → 内存缓存 → 磁盘缓存 → 联网 → 解析 → Ready/Empty/Failed。
 * 失败**不影响播放**——歌词是可选层。
 *
 * 联网恢复后（[connectivity] 由 false 变 true）若当前处于 Failed，自动重试一次。
 */
class LyricViewModel(
    private val provider: MusicProvider,
    private val memoryCache: com.musicbox.app.lyric.LyricCache = sharedLyricCache,
    private val diskCache: LyricDiskCache? = null,
    private val connectivity: kotlinx.coroutines.flow.Flow<Boolean>? = null,
) : ViewModel() {

    private val _state = MutableStateFlow<LyricUiState>(LyricUiState.Idle)
    val state: StateFlow<LyricUiState> = _state.asStateFlow()

    private val _showTranslation = MutableStateFlow(true)
    val showTranslation: StateFlow<Boolean> = _showTranslation.asStateFlow()

    private val _showRomaji = MutableStateFlow(false)
    val showRomaji: StateFlow<Boolean> = _showRomaji.asStateFlow()

    private var loadedSongId: Long? = null
    private var job: Job? = null

    init {
        connectivity?.let { flow ->
            viewModelScope.launch {
                flow.distinctUntilChanged().collect { online ->
                    val id = loadedSongId ?: return@collect
                    if (online && _state.value is LyricUiState.Failed) {
                        startLoad(id, useMemoryCache = false)
                    }
                }
            }
        }
    }

    fun toggleTranslation() { _showTranslation.value = !_showTranslation.value }
    fun toggleRomaji() { _showRomaji.value = !_showRomaji.value }

    fun setShowTranslation(value: Boolean) { _showTranslation.value = value }
    fun setShowRomaji(value: Boolean) { _showRomaji.value = value }

    /** 歌词页"重试"按钮：忽略缓存重新拉。 */
    fun retry() {
        loadedSongId?.let { startLoad(it, useMemoryCache = false) }
    }

    /** 切歌时调用。同一首重复调用直接忽略；null 表示无当前歌 → Idle。 */
    fun load(songId: Long?) {
        if (songId == null) {
            job?.cancel()
            loadedSongId = null
            _state.value = LyricUiState.Idle
            return
        }
        if (songId == loadedSongId) return
        startLoad(songId, useMemoryCache = true)
    }

    private fun startLoad(songId: Long, useMemoryCache: Boolean) {
        loadedSongId = songId
        job?.cancel()

        if (useMemoryCache) {
            memoryCache.get(songId)?.let {
                _state.value = toReady(it)
                return
            }
            diskCache?.get(songId)?.let { raw ->
                val timeline = LyricParser.parse(raw)
                memoryCache.put(songId, timeline)
                _state.value = toReady(timeline)
                return
            }
        }

        _state.value = LyricUiState.Loading
        job = viewModelScope.launch {
            val result = provider.getLyric(songId)
            if (loadedSongId != songId) return@launch
            val data = result.getOrNull()
            if (data == null) {
                _state.value = LyricUiState.Failed
                return@launch
            }
            // 有内容才落盘：避免把"这首歌其实没歌词"也缓存成文件
            if (listOfNotNull(data.yrc, data.lrc, data.translation, data.romaji).any { it.isNotBlank() }) {
                diskCache?.put(songId, data)
            }
            val timeline = withContext(Dispatchers.Default) { LyricParser.parse(data) }
            if (loadedSongId != songId) return@launch

        }
    }
```

```kotlin
/** /eapi/login/qrcode/unikey 的响应。 */

    @SerializedName("unikey") val unikey: String? = null,

    @SerializedName("cookie") val cookie: String? = null,
)

/** /eapi/nuser/account/get 的响应。 */

    @SerializedName("code") val code: Int = -1,

)

internal data class ProfileDto(
    @SerializedName("userId") val userId: Long = 0L,
    @SerializedName("nickname") val nickname: String? = null,
)

/** 登录 cookie 串 → 键值对。形如 `MUSIC_U=xxx; __csrf=yyy; NMTID=zzz`。 */
internal fun parseCookieString(cookie: String): Map<String, String> =
    cookie.split(";")
        .mapNotNull { part ->

            if (idx <= 0) null else part.substring(0, idx).trim() to part.substring(idx + 1).trim()
        }
```

```kotlin
package com.musicbox.app.source.netease

import com.google.gson.Gson
import com.google.gson.JsonSyntaxException
import com.musicbox.app.source.SourceError
import java.io.IOException
import java.net.URLEncoder

/** 轮询结果。语义见计划 §1.1。 */
sealed interface QrPollResult {
    /** 801：等待用户扫码。 */
    data object Waiting : QrPollResult
    /** 800：二维码已过期，需要重新取 key。 */
    data object Expired : QrPollResult
    /** 802：已扫码，等待用户在手机上确认。 */
    data object Scanned : QrPollResult
    /** 803：登录成功，携带 MUSIC_U / __csrf。 */
    data class Success(val musicU: String, val csrf: String) : QrPollResult
}

/**
 * 扫码登录接口。**复用 eapi 签名**，不需要 weapi / RSA（实测结论见计划 §1.1）。
 * 与 NetEaseProvider 一样只做"组 JSON → 签名 → 发请求 → 解析"。
 */
class NetEaseLoginApi(
    private val transport: HttpTransport,
    private val deviceId: String,
    private val gson: Gson = Gson(),
) {

    /** 取二维码 key。二维码内容用 `https://music.163.com/login?codekey=<unikey>`。 */
    fun createQrKey(): Result<String> = call(
        uri = UNIKEY_URI,
        payload = linkedMapOf<String, Any>("type" to 1, "e_r" to false, "header" to NetEaseHeader.build(deviceId)),
    ) { body ->
        val dto = gson.fromJson(body, QrUnikeyRespDto::class.java)
        val key = dto?.unikey
        if (dto?.code != 200 || key.isNullOrBlank()) {
            throw SourceError.Api(dto?.code ?: -1, null)
        }
        key
    }

    /** 轮询扫码状态。调用方每 2 秒轮一次。 */
    fun poll(unikey: String): Result<QrPollResult> = call(
        uri = POLL_URI,
        payload = linkedMapOf<String, Any>(
            "key" to unikey,
            "type" to 1,
            "e_r" to false,
            "header" to NetEaseHeader.build(deviceId),
        ),
    ) { body ->
        val dto = gson.fromJson(body, QrPollRespDto::class.java)
        when (dto?.code) {
            801 -> QrPollResult.Waiting
            800 -> QrPollResult.Expired
            802 -> QrPollResult.Scanned
            803 -> {
                val cookie = parseCookieString(dto.cookie.orEmpty())
                val musicU = cookie["MUSIC_U"]
                    ?: throw SourceError.Parse("登录成功但响应缺少 MUSIC_U", null)
                QrPollResult.Success(musicU = musicU, csrf = cookie["__csrf"].orEmpty())
            }
            else -> throw SourceError.Api(dto?.code ?: -1, null)
        }
    }

    /** 取昵称（可选增强）。失败不影响登录态。 */
    fun fetchNickname(musicU: String, csrf: String): Result<String?> = call(
        uri = ACCOUNT_URI,
        payload = linkedMapOf<String, Any>("e_r" to false, "header" to NetEaseHeader.build(deviceId, csrf)),
        musicU = musicU,
    ) { body ->
        gson.fromJson(body, AccountRespDto::class.java)?.profile?.nickname
    }

    /** 统一封装：构造 header → 签名 → post → 解析。bodyMapper 里抛 SourceError 即失败。 */
    private fun <T> call(
        uri: String,
        payload: Map<String, Any>,
        musicU: String? = null,
        bodyMapper: (String) -> T,
    ): Result<T> {
        return try {
            val header = payload["header"] as Map<String, String>
            val params = NetEaseCrypto.eapiParams(uri, gson.toJson(payload))
            val body = "params=" + URLEncoder.encode(params, "UTF-8")
            val headers = mapOf(
                "User-Agent" to NetEaseHeader.UA,
                "Cookie" to NetEaseHeader.cookieHeader(header, musicU),
                "Referer" to "https://music.163.com",
            )
            // uri 形如 /api/login/qrcode/unikey → /eapi/login/qrcode/unikey
            val response = transport.post(EAPI_BASE + uri.replaceFirst("/api", "/eapi"), body, headers)
            Result.success(bodyMapper(response.body))
        } catch (e: SourceError) {
            Result.failure(e)
        } catch (e: JsonSyntaxException) {
            Result.failure(SourceError.Parse("响应不是合法 JSON", e))
        } catch (e: IOException) {
            Result.failure(SourceError.Network(e))
        }
    }

    private companion object {
        const val EAPI_BASE = "https://interfacepc.music.163.com"
        const val UNIKEY_URI = "/api/login/qrcode/unikey"
        const val POLL_URI = "/api/login/qrcode/client/login"
        const val ACCOUNT_URI = "/api/nuser/account/get"
    }
}
```

```kotlin
     * 由 header 对象拼 Cookie。登录后额外带上 MUSIC_U（网易云鉴权的核心 cookie），
```

```kotlin
package com.musicbox.app.ui.login

import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import com.musicbox.app.data.SessionStore
import com.musicbox.app.source.netease.NetEaseLoginApi
import com.musicbox.app.source.netease.QrPollResult
import kotlinx.coroutines.Job
import kotlinx.coroutines.delay
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.isActive
import kotlinx.coroutines.launch

/** 扫码登录界面状态。 */
sealed interface LoginUiState {
    data object Idle : LoginUiState
    /** 正在申请二维码。 */
    data object Loading : LoginUiState
    /** 二维码已就绪，等待扫码。 */
    data class Waiting(val qrContent: String) : LoginUiState
    /** 已扫码，等待手机确认。 */
    data object Scanned : LoginUiState
    /** 二维码过期，可点"刷新"。 */
    data object Expired : LoginUiState
    /** 登录成功。 */
    data class Success(val nickname: String?) : LoginUiState
    /** 出错（网络/接口）。 */
    data class Failed(val message: String) : LoginUiState
}

/**
 * 扫码登录状态机。
 * 流程：createQrKey → Waiting → 每 2s 轮询 → Scanned → Success。
 * 二维码过期（800）→ 自动重新取 key。轮询失败不清空二维码，等下一次轮询自愈。
 */
class LoginViewModel(
    private val api: NetEaseLoginApi,
    private val sessionStore: SessionStore,
    private val pollIntervalMs: Long = 2_000L,
) : ViewModel() {

    private val _state = MutableStateFlow<LoginUiState>(LoginUiState.Idle)
    val state: StateFlow<LoginUiState> = _state.asStateFlow()

    private var pollJob: Job? = null

    /** 打开登录页时调用。幂等：已在等扫码就不重复申请。 */
    fun start() {
        if (_state.value is LoginUiState.Waiting || _state.value is LoginUiState.Scanned) return
        if (sessionStore.session.value.isLoggedIn) {
            _state.value = LoginUiState.Success(sessionStore.session.value.nickname)
            return
        }
        requestQrKey()
    }

    /** "刷新二维码"。 */
    fun refresh() {
        pollJob?.cancel()
        requestQrKey()
    }

    fun reset() {
        pollJob?.cancel()
        _state.value = LoginUiState.Idle
    }

    private fun requestQrKey() {
        pollJob?.cancel()
        _state.value = LoginUiState.Loading
        pollJob = viewModelScope.launch {
            api.createQrKey()
                .onSuccess { key ->
                    _state.value = LoginUiState.Waiting(QR_PREFIX + key)
                    pollLoop(key)
                }
                .onFailure { _state.value = LoginUiState.Failed("二维码获取失败，请检查网络") }
        }
    }

    private suspend fun pollLoop(unikey: String) {
        while (viewModelScope.isActive) {
            delay(pollIntervalMs)
            val result = api.poll(unikey)
            when {
                result.isFailure -> Unit // 网络抖动：保留二维码，下一轮再试
                else -> when (val r = result.getOrThrow()) {
                    QrPollResult.Waiting -> Unit
                    QrPollResult.Scanned -> _state.value = LoginUiState.Scanned
                    QrPollResult.Expired -> {
                        requestQrKey() // 过期 → 自动重取（会 cancel 本 job）
                        return
                    }
                    is QrPollResult.Success -> {
                        sessionStore.save(r.musicU, r.csrf, null)
                        _state.value = LoginUiState.Success(null)
                        // 昵称是可选增强，单独再取一次；失败不影响登录成功
                        api.fetchNickname(r.musicU, r.csrf).getOrNull()?.let { nick ->
                            sessionStore.updateNickname(nick)
                            _state.value = LoginUiState.Success(nick)
                        }
                        return
                    }
                }
            }
        }
    }

    override fun onCleared() {
        pollJob?.cancel()
        super.onCleared()
    }

    private companion object {
        const val QR_PREFIX = "https://music.163.com/login?codekey="
    }
}
```

```kotlin
/**
 * 扫码登录页。二维码**本地用 ZXing 生成位图**（纯原生，不加载任何网页）。
 * 轮询在 LoginViewModel 里每 2 秒跑一次。
 */
@Composable
fun LoginScreen(
    viewModel: LoginViewModel,
    onLoggedIn: () -> Unit,
    onBack: () -> Unit,
) {
    val state by viewModel.state.collectAsStateWithLifecycle()

    androidx.compose.runtime.LaunchedEffect(state) {
        if (state is LoginUiState.Success) onLoggedIn()
    }

    Column(
        modifier = Modifier.fillMaxSize().padding(24.dp),
        horizontalAlignment = Alignment.CenterHorizontally,
        verticalArrangement = Arrangement.Center,
    ) {
        Text("扫码登录网易云", style = MaterialTheme.typography.titleLarge)
        Text(
            text = "登录后可解锁完整播放与更高音质；不登录也能正常搜索与试听。",
            style = MaterialTheme.typography.bodySmall,
            color = MaterialTheme.colorScheme.onSurfaceVariant,
            modifier = Modifier.padding(top = 8.dp, bottom = 24.dp),
        )

        when (val s = state) {
            is LoginUiState.Loading, LoginUiState.Idle -> CircularProgressIndicator()
            is LoginUiState.Waiting -> {
                QrImage(s.qrContent)
                Text("打开网易云音乐 App → 扫一扫", modifier = Modifier.padding(top = 16.dp))
            }
            LoginUiState.Scanned -> {
                CircularProgressIndicator()
                Text("已扫码，请在手机上确认", modifier = Modifier.padding(top = 16.dp))
            }
            LoginUiState.Expired -> {
                Text("二维码已过期")
                TextButton(onClick = viewModel::refresh) { Text("刷新二维码") }
            }
            is LoginUiState.Failed -> {
                Text(s.message, color = MaterialTheme.colorScheme.error)
                TextButton(onClick = viewModel::refresh) { Text("重试") }
            }
            is LoginUiState.Success -> Text("登录成功")
        }

        TextButton(onClick = onBack, modifier = Modifier.padding(top = 24.dp)) { Text("返回") }
    }
}

@Composable
private fun QrImage(content: String) {
    val bitmap = remember(content) { encodeQr(content) }
    Box(modifier = Modifier.size(220.dp), contentAlignment = Alignment.Center) {
        if (bitmap == null) {
            Text("二维码生成失败", color = MaterialTheme.colorScheme.error)
        } else {
            Image(bitmap = bitmap.asImageBitmap(), contentDescription = "登录二维码")
        }
    }
}

/**
 * 用 ZXing 把文本编码成二维码位图。纯 CPU，不联网。
 * 尺寸取 640px，够任何屏幕清晰显示。
 */
private fun encodeQr(content: String): android.graphics.Bitmap? = runCatching {
    val size = 640
    val hints = mapOf(
        EncodeHintType.CHARACTER_SET to "UTF-8",
        EncodeHintType.ERROR_CORRECTION to ErrorCorrectionLevel.M,
        EncodeHintType.MARGIN to 1,
    )
    val matrix = QRCodeWriter().encode(content, BarcodeFormat.QR_CODE, size, size, hints)
    val pixels = IntArray(size * size)
    for (y in 0 until size) {
        for (x in 0 until size) {
            pixels[y * size + x] = if (matrix[x, y]) BLACK else WHITE
        }
    }
    android.graphics.Bitmap.createBitmap(pixels, size, size, android.graphics.Bitmap.Config.ARGB_8888)
}.getOrNull()

private const val BLACK = 0xFF000000.toInt()
private const val WHITE = 0xFFFFFFFF.toInt()
```

```kotlin
package com.musicbox.app.player

import android.content.Context
import android.net.ConnectivityManager
import android.net.Network
import android.net.NetworkCapabilities
import kotlinx.coroutines.channels.awaitClose
import kotlinx.coroutines.flow.Flow
import kotlinx.coroutines.flow.callbackFlow
import kotlinx.coroutines.flow.distinctUntilChanged

/**
 * 在线状态监测（设计文档 §9：网络中断 → 缓冲指示 + 自动重连提示）。
 * 用「已验证可上网」（VALIDATED）判定，避免连上热点但没网时误判为在线。
 */
class NetworkMonitor(context: Context) {

    private val cm = context.getSystemService(Context.CONNECTIVITY_SERVICE) as ConnectivityManager

    fun isOnline(): Boolean {
        val network = cm.activeNetwork ?: return false
        val caps = cm.getNetworkCapabilities(network) ?: return false
        return caps.hasCapability(NetworkCapabilities.NET_CAPABILITY_INTERNET) &&
            caps.hasCapability(NetworkCapabilities.NET_CAPABILITY_VALIDATED)
    }

    /** 在线状态变化流（首帧即当前值）。 */
    fun observe(): Flow<Boolean> = callbackFlow {
        trySend(isOnline())
        val callback = object : ConnectivityManager.NetworkCallback() {
            override fun onAvailable(network: Network) { trySend(true) }
            override fun onLost(network: Network) { trySend(isOnline()) }
            override fun onCapabilitiesChanged(network: Network, caps: NetworkCapabilities) {
                trySend(
                    caps.hasCapability(NetworkCapabilities.NET_CAPABILITY_INTERNET) &&
                        caps.hasCapability(NetworkCapabilities.NET_CAPABILITY_VALIDATED),
                )
            }
        }
        cm.registerDefaultNetworkCallback(callback)
        awaitClose { cm.unregisterNetworkCallback(callback) }
    }.distinctUntilChanged()
}
```

```kotlin
     * 取流失败自动重连（设计文档 §9：指数退避，最多 3 次）。
     * 退避 0.5s → 1s → 2s，总等待 3.5s，不会让用户等太久，也足以扛过瞬时抖动。
     */
    private suspend fun resolveWithRetry(songId: Long): com.musicbox.app.model.StreamInfo? {
        var backoff = 500L
        repeat(MAX_STREAM_ATTEMPTS) { attempt ->
            val info = withTimeoutOrNull(RESOLVE_TIMEOUT_MS) {
                container.provider.getStreamUrl(songId)
            }?.getOrNull()
            if (info != null) return info
            if (attempt < MAX_STREAM_ATTEMPTS - 1) {
                kotlinx.coroutines.delay(backoff)
                backoff *= 2
            }
        }
        return null
    }
```

```kotlin
package com.musicbox.app.source.failover

import com.google.gson.Gson
import com.google.gson.JsonSyntaxException
import com.musicbox.app.model.LyricData
import com.musicbox.app.model.SearchType
import com.musicbox.app.model.Song
import com.musicbox.app.model.StreamInfo
import com.musicbox.app.source.MusicProvider
import com.musicbox.app.source.SourceError
import com.musicbox.app.source.netease.HttpTransport
import com.musicbox.app.source.netease.LyricRespDto
import com.musicbox.app.source.netease.SearchRespDto
import com.musicbox.app.source.netease.SongUrlRespDto
import com.musicbox.app.source.netease.TransportResponse
import com.musicbox.app.source.netease.toSong
import com.musicbox.app.source.netease.toStreamInfo
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import java.io.IOException
import java.net.URLEncoder

/**
 * 备用音源（设计文档 §7）：一个**地址可在设置中配置**的第三方公共接口实例。
 * 默认地址留空 = 未配置，不参与正常流程；只在端上签名路径临时失效时作为应急兜底。
 *
 * 协议与 NetEaseCloudMusicApi 一致，响应结构与 NetEaseDto 完全兼容 → 直接复用 DTO。
 * 首选路径永远是端上自包含（NetEaseProvider），本类不参与正常流程。
 */
class FallbackProvider(
    private val transport: HttpTransport,
    private val baseUrl: () -> String,
    private val gson: Gson = Gson(),
) : MusicProvider {

    override val id: String = "fallback"

    private fun base(): String = baseUrl().trim().trimEnd('/')

    override suspend fun search(
        keyword: String,
        type: SearchType,
        page: Int,
    ): Result<List<Song>> = withContext(Dispatchers.IO) {
        val base = base()
        if (base.isEmpty()) return@withContext Result.failure(SourceError.NotConfigured())
        try {
            val limit = PAGE_SIZE
            val offset = (page - 1).coerceAtLeast(0) * limit
            val url = "$base/search?keywords=${enc(keyword)}&type=${type.apiValue}" +
                "&limit=$limit&offset=$offset"
            val body = transport.get(url, HEADERS).body
            val dto = parse(body, SearchRespDto::class.java)
            if (dto.code != 200) throw SourceError.Api(dto.code, null)
            Result.success(dto.result?.songs.orEmpty().map { it.toSong(id) })
        } catch (e: SourceError) {
            Result.failure(e)
        } catch (e: IOException) {
            Result.failure(SourceError.Network(e))
        }
    }

    override suspend fun getStreamUrl(songId: Long): Result<StreamInfo> = withContext(Dispatchers.IO) {
        val base = base()
        if (base.isEmpty()) return@withContext Result.failure(SourceError.NotConfigured())
        try {
            val url = "$base/song/url/v1?id=$songId&level=exhigh"
            val body = transport.get(url, HEADERS).body
            val dto = parse(body, SongUrlRespDto::class.java)
            if (dto.code != 200) throw SourceError.Api(dto.code, null)
            val info = dto.data?.firstOrNull()?.toStreamInfo()
                ?: throw SourceError.Unplayable("当前歌曲暂无可用播放地址")
            Result.success(info)
        } catch (e: SourceError) {
            Result.failure(e)
        } catch (e: IOException) {
            Result.failure(SourceError.Network(e))
        }
    }

    override suspend fun getLyric(songId: Long): Result<LyricData> = withContext(Dispatchers.IO) {
        val base = base()
        if (base.isEmpty()) return@withContext Result.failure(SourceError.NotConfigured())
        try {
            val body = transport.get("$base/lyric?id=$songId", HEADERS).body
            val dto = parse(body, LyricRespDto::class.java)
            if (dto.code != 200) throw SourceError.Api(dto.code, null)
            Result.success(
                LyricData(
                    yrc = dto.yrc?.lyric?.takeIf { it.isNotBlank() },
                    lrc = dto.lrc?.lyric?.takeIf { it.isNotBlank() },
                    translation = dto.tlyric?.lyric?.takeIf { it.isNotBlank() },
                    romaji = dto.romalrc?.lyric?.takeIf { it.isNotBlank() },
                ),
            )
        } catch (e: SourceError) {
            Result.failure(e)
        } catch (e: IOException) {
            Result.failure(SourceError.Network(e))
        }
    }

    private fun <T> parse(body: String, clazz: Class<T>): T =
        try {
            gson.fromJson(body, clazz) ?: throw SourceError.Parse("响应为空", null)
        } catch (e: JsonSyntaxException) {
            throw SourceError.Parse("响应不是合法 JSON", e)
        }

    private fun enc(value: String): String = URLEncoder.encode(value, "UTF-8")

    private companion object {
        const val PAGE_SIZE = 30
        val HEADERS = mapOf("User-Agent" to "musicbox/1.0", "Accept" to "application/json")
    }
}
```

```kotlin
package com.musicbox.app.source

import com.musicbox.app.model.LyricData
import com.musicbox.app.model.SearchType
import com.musicbox.app.model.Song
import com.musicbox.app.model.StreamInfo

/**
 * 主源 + 备用源的自动降级（设计文档 §9：接口失效 → 自动切备用源重试一次）。
 *
 * 只在**可重试类错误**（网络 / 接口 code / 解析）上试备用源；
 * `Unplayable`（无版权、需登录）与 `NotConfigured` 属于"确定不可播"，不再浪费一次请求。
 * 备用源也失败时，**保留主源错误**——它更贴近真实原因。
 */
class FailoverProvider(
    private val primary: MusicProvider,
    private val fallback: MusicProvider,
) : MusicProvider {

    override val id: String = primary.id

    override suspend fun search(
        keyword: String,
        type: SearchType,
        page: Int,
    ): Result<List<Song>> = degrade({ primary.search(keyword, type, page) }) {
        fallback.search(keyword, type, page)
    }

    override suspend fun getStreamUrl(songId: Long): Result<StreamInfo> =
        degrade({ primary.getStreamUrl(songId) }) { fallback.getStreamUrl(songId) }

    override suspend fun getLyric(songId: Long): Result<LyricData> =
        degrade({ primary.getLyric(songId) }) { fallback.getLyric(songId) }

    private suspend fun <T> degrade(
        primaryCall: suspend () -> Result<T>,
        fallbackCall: suspend () -> Result<T>,
    ): Result<T> {
        val primaryResult = primaryCall()
        if (primaryResult.isSuccess) return primaryResult
        val error = primaryResult.exceptionOrNull()
        if (!isRetryable(error)) return primaryResult
        val fallbackResult = fallbackCall()
        return if (fallbackResult.isSuccess) fallbackResult else primaryResult
    }

    private fun isRetryable(error: Throwable?): Boolean = when (error) {
        is SourceError.Network, is SourceError.Api, is SourceError.Parse -> true
        else -> false
    }
}
```

```kotlin
package com.musicbox.app.ui.search

import com.musicbox.app.model.Song

/**
 * 查询词与搜索结果的相关性判定（纯函数，可单测）。
 *
 * 动机（MI-1 实测遗留）：网易云 cloudsearch/pc 对乱码 / 无意义关键词**不会返回空列表**，
 * 而是给一批模糊匹配结果。所以"结果非空"不等于"搜到了"，需要额外判定，
 * 否则用户搜 `asdkjhqwe` 会看到一堆不相干的歌，空状态永远触发不了。
 *
 * 判定规则：把查询词按空白切成词元，只要**任一结果**的歌名或歌手包含**任一词元**
 * （不区分大小写）即视为"有真实匹配"。查询词没有词元（纯空白）时视为有匹配（交给正常流程）。
 */
fun hasRealMatch(query: String, songs: List<Song>): Boolean {
    val tokens = query.trim().lowercase().split(Regex("\\s+")).filter { it.isNotEmpty() }
    if (tokens.isEmpty()) return true
    return songs.any { song ->
        val haystack = (song.name + " " + song.artists.joinToString(" ")).lowercase()
        tokens.any { haystack.contains(it) }
    }
}
```
